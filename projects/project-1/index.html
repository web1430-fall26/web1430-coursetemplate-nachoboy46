                        export function renderMtgCard(_cardName,_image,_cardType,_text,_cost=null){
                            const mtgCard = document.createElement('article');
                            mtgCard.className = "mtg-card";
                            mtgCard.setAttribute("aria-hidden", "false");

                            const cardHeader = document.createElement("h3");
                            cardHeader.className = "mtg-card-name"
                            cardHeader.textContent = _cardName;
                            mtgCard.appendChild(cardHeader);

                            const cardArtSpacer = document.createElement("div");
                            const cardImage = document.createElement("img");
                            cardArtSpacer.appendChild(cardImage);
                            cardImage.className = "mtg-card-art";
                            cardImage.src = _image;
                            cardImage.alt = ("Image of the "+_cardName+" magic the gathering card");
                            mtgCard.appendChild(cardArtSpacer);

                            const hiddenSection = document.createElement("div");
                            hiddenSection.className="mtg-card-hidden-section";
                            hiddenSection.hidden = true;
                            hiddenSection.setAttribute('aria-hidden','true');
                            
                            mtgCard.appendChild(hiddenSection);

                            if(_cost){
                                const cardCostSpacer = document.createElement("div");
                                const cardCost = document.createElement("span");
                                cardCostSpacer.appendChild(cardCost);
                                cardCost.className = "mtg-card-cost"
                                cardCost.textContent = _cost;
                                hiddenSection.appendChild(cardCostSpacer);
                            }
                            
                            const cardTypeSpacer = document.createElement("div");
                            const cardType = document.createElement("span");
                            cardTypeSpacer.appendChild(cardType);
                            cardType.className = "mtg-card-type"
                            cardType.textContent = _cardType;
                            hiddenSection.appendChild(cardTypeSpacer);

                            const cardTextSpacer = document.createElement('div');
                            const cardText = document.createElement('p');
                            cardTextSpacer.appendChild(cardText);
                            cardText.className = "mtg-card-text"
                            cardText.textContent = _text;
                            hiddenSection.appendChild(cardTextSpacer);

                            const button = document.createElement('button');
                            button.className = "mtg-card-show-more-button";
                            button.textContent = 'show more';
                            mtgCard.appendChild(button);


                            return mtgCard;
                        }

                        function initMtgCardButtons(cardsSection){
                            cardsSection.addEventListener('click',function(event){
                                if(event.target.matches('.mtg-card-show-more-button')){
                                    const card = event.target.closest('.mtg-card');
                                    const hiddenSection = card.querySelector('.mtg-card-hidden-section');
                                    if(hiddenSection.getAttribute("aria-hidden") === "true"){
                                        hiddenSection.hidden = false;
                                        hiddenSection.setAttribute('aria-hidden', "false");
                                    } else {
                                        hiddenSection.hidden = true;
                                        hiddenSection.setAttribute('aria-hidden', "true");
                                    }
                                }
                            });
                        }

                        function searchCards(_searchTerm = null){
                            const cards = document.querySelectorAll(".mtg-card");
                            const searchBar = document.getElementById("search-bar");

                            if(_searchTerm === ""){
                                cards.forEach(function (card){
                                    card.style.display = "block";
                                    card.setAttribute("aria-hidden","false");
                                    return;
                                });
                            }

                            let searchTerm = null;
                            if(!_searchTerm){
                                searchTerm = searchBar.value.trim().toLowerCase();
                            } else {
                                searchTerm = _searchTerm;
                            }
                            
                            
                            cards.forEach(function (card){
                                const nameElement = card.querySelector(".mtg-card-name");
                                const cardName = nameElement.textContent.toLocaleLowerCase();
                                if(cardName.includes(searchTerm)){
                                    card.style.display = "block";
                                    card.setAttribute("aria-hidden","false");
                                } else {
                                    card.style.display = "none";
                                    card.setAttribute("aria-hidden","true");
                                }
                            });

                            getValidCards();
                        }

                        function initSearchBar(searchBar){
                            const searchForm = document.getElementById("search-form");
                            searchForm.addEventListener("submit", function (event){
                                event.preventDefault();
                            });
                            searchBar.addEventListener('input', function (event){
                                event.preventDefault();
                                searchCards();
                            });
                        }

Magic the Gathering Phases

To start the beginning phase , you untap (turn to face right side up) all permanents (Things laid out in front of you) that you control. Then, you have your upkeep. This doesn't do anything by itself, but some cards have triggers here, so keep note of it. Lastly, you draw a card.

During your first main phase, you can cast spells and play lands. You can play one land per turn by taking a land card from your hand and playing it down in front of you. You also play the majority of your spells during your main phase, as most spells (with a few exceptions, like instants or creatures with flash) are cast during you main phase. You may cast as many spells as you have mana to pay for. You get mana by tapping your lands. (turning them sideways.)

Combat starts with the beginning of phase. This is similar to your upkeep, in that stuff only happens if a card references this. Then, you enter your declaring attackers phase. You declare which of your creatures in front of you are attacking and what player or planeswalker they are attacking. Them you tap them (turn them sideways) to show this. After this, there is the declare blockers phase. Your opponent looks at the untapped creatures in front of them, and declares which of them, if any, they would like to have block the creatures you declared attacking. Then, there is the damage step, in which creatures deal damage to eachother and players, potentially killing creatuers and or players. Then, there is the end of combat step, which functions like the beginning of combat step.

The second main phase behaves exactly like the first main phase. You can cast more spells, and you can also play a land card if you still haven't played on yet this turn. The only difference is this one is after combat.

The phase in your turn is the end step. There is first the beginning of your end step, which behaves like upkeep and has the chance to trigger certain abilities based on the board state, but by itself does nothing. Then there is the discard step, in which your will discard down to the maximum hand size of seven cards in the event that you have more than seven cards in hand. After this your turn is over and the next player in the turn rotation takes their turn.
